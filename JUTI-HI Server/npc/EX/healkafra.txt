morocc03,150,192,6	script	Kafra::eternalmid	114,{
	getmapxy .@mapname$,.@mapx,.@mapy,BL_PC;
	cutin "kafra_04",2;
	callfunc "F_KafSet";
	mes "[Kafra Employee]";
	mes "ยินดีต้อนรับคุณ ^FF0000"+strcharinfo(0)+"^000000";
	mes "Char ID ของคุณคือ ^FF0000"+getcharid(0)+"^000000";
	if(getcharid(1) > 0){
		mes "Party ID ของคุณคือ ^FF0000"+getcharid(1)+"^000000";
	}else{
		mes "^FF0000คุณยังไม่ได้เข้าร่วม Party^000000";
	}
	if(getcharid(2) > 0){
		mes "Guild ID ของคุณคือ ^FF0000"+getcharid(2)+"^000000";
	}else{
		mes "^FF0000คุณยังไม่ได้สังกัด Guild^000000";
	}
	mes "กรุณาเลือกบริการที่ต้องการ.";
	mes "พนักงาน Kafra พร้อมให้บริการคุณ";
	callfunc "F_Kafra",5,7,0,60,930;
	savepoint "morocc03",.@mapx,.@mapy,1,1;
	callfunc "F_KafEnd",0,1,"เมือง JUTI";
	OnInit:
	waitingroom "บริการ Kafra",0;
	end;
}
