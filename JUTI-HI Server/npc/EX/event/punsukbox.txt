function	script	F_punsukbox	{
//ตั้งจำนวนทั้งหมด
.@MM500 = 10;	//บัตรแลกเงินสด 500บาท
.@MM100 = 50;	//บัตรแลกเงินสด 100บาท
.@VC500 = 10;	//Gift Voucher เติมเงิน 500บาท
.@VC100 = 50;	//Gift Voucher เติมเงิน 100บาท

.@rand	= rand(1,1000);
if(.@rand == 779 && $punsukMM500 < 10){
getitem 3130035,1;
$punsukMM500 += 1;
announce "[ PunSuk Box ] : ผู้เล่น ["+strcharinfo(0)+"] ได้รับ บัตรแลกเงินสด 500บาท ( "+$punsukMM500+"/"+.@MM500+" )",bc_all,0xFF8000;
}
else if (.@rand == 277 && $punsukMM100 < 50){
getitem 3130038,1;
$punsukMM100 += 1;
announce "[ PunSuk Box ] : ผู้เล่น ["+strcharinfo(0)+"] ได้รับ บัตรแลกเงินสด 100บาท ( "+$punsukMM100+"/"+.@MM100+" )",bc_all,0xFF8000;
}
else if (.@rand == 377 && $punsukVC500 < 10){
getitem 3130039,1;
$punsukVC500 += 1;
announce "[ PunSuk Box ] : ผู้เล่น ["+strcharinfo(0)+"] ได้รับ Gift Voucher เติมเงิน 500บาท ( "+$punsukVC500+"/"+.@VC500+" )",bc_all,0xFF8000;
}
else if (.@rand == 877 && $punsukVC100 < 50){
getitem 3130040,1;
$punsukVC100 += 1;
announce "[ PunSuk Box ] : ผู้เล่น ["+strcharinfo(0)+"] ได้รับ Gift Voucher เติมเงิน 100บาท ( "+$punsukVC100+"/"+.@VC100+" )",bc_all,0xFF8000;
}
else{
.@randzeny = rand(500,7000);
zeny += .@randzeny;
dispbottom "[ PunSuk Box ] : คุณได้รับรางวัลปลอบใจ " + callfunc("F_InsertComma",.@randzeny) + " Zeny",0xAEFF00;
end;
}
}
