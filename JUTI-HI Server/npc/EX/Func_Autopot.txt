function	script	autopot	{
	while(1){
		mes " [Auto-Potion]";
		mes (@autopotion_on)?"สถานะ : ^0000FFเปิด^000000":"สถานะ : ^FF0000ปิด^000000";
		mes (AP_HP_RATE)?"HP : "+AP_HP_RATE+" %":"HP : ยังไม่ได้ตั้งค่า";
		mes (AP_SP_RATE)?"SP : "+AP_SP_RATE+" %":"SP : ยังไม่ได้ตั้งค่า";
		
		mes " รายการยา";
		if(AP_HP_ITEMID)
			mes "HP :"+getitemname(AP_HP_ITEMID)+" ("+countitem(AP_HP_ITEMID)+" ea)";
		else
			mes "HP : ยังไม่ได้เลือกไอเทม";
			
		if(AP_SP_ITEMID)
			mes "SP :"+getitemname(AP_SP_ITEMID)+" ("+countitem(AP_SP_ITEMID)+" ea)";
		else
			mes "SP : ยังไม่ได้เลือกไอเทม";
		
		switch(select("เปิด/ปิด ","ตั้งค่า HP","ตั้งค่า SP","ตั้งไอเทม HP","ตั้งไอเทม SP","ออก")){
			case 1:
				if(!@autopotion_on){
					//validate
					if(!AP_HP_ITEMID || !AP_SP_ITEMID || !AP_HP_RATE || !AP_SP_RATE){
						next;
						mes "[Auto-Potion]";
						mes "กรุณาตั้งค่าให้ครบถ้วน";
					}else{
						next;
						mes "[Auto-Potion]";
						set @autopotion_on,1;
						mes "เปิดเรียบร้อย";
						deltimer "start_autopot::OnStart";
							if (getcharid(0)==2470 ||getcharid(0)==2425 ||getcharid(0)==2425 ||getcharid(0)==2423 ||getcharid(0)==2430 || getcharid(0)==2343 ||getcharid(0)==2429 ||getcharid(0)==2428 ||getcharid(0)==2394 ||getcharid(0) == 2396 ||getcharid(0) == 2426 ||getcharid(0) == 2424 ||getcharid(0) == 2351 ||getcharid(0) == 2438 ||getcharid(0) == 2420 ||getcharid(0) == 2432 ||getcharid(0) == 1815 ||getcharid(0) == 2270 ||getcharid(0) == 1897 ||getcharid(0) == 1823 ||getcharid(0) == 1425 ||getcharid(0) == 1407 ||getcharid(0) == 408 ||getcharid(0) == 2200 ||getcharid(0) == 1606 ||getcharid(0) == 1590 ||getcharid(0) == 2132 ||getcharid(0) == 1816 ||getcharid(0) == 1849 ||getcharid(0) == 1971 || getcharid(0)== 1290 ||getcharid(0)==1609 || getcharid(0)==700 || getcharid(0)==1880 || getcharid(0)==1879 || getcharid(0)==1645 || getcharid(0)==1469 || getcharid(0)==731 || getcharid(0)==1404 || getcharid(0)==58|| getcharid(0)==1154|| getcharid(0)==1582) { 		
								addtimer 100,"start_autopot::OnStart";
							}else{
								addtimer $@AUTOPOT_DELAY,"start_autopot::OnStart";
							}	
							//addtimer $@AUTOPOT_DELAY,"start_autopot::OnStart";
						end;
					}
				}else{
					set @autopotion_on,0;
					mes "ปิดเรียบร้อย";
					deltimer "start_autopot::OnStart";
					end;
				}
				
				next;
				break;
			case 2:
				next;
				mes "[Auto-Potion]";
				mes "กรุณาใส่ HP % ที่ต้องการให้ทำงาน";
				mes "";
				input .@rate,0,100;
				set AP_HP_RATE,.@rate;
				if(.@rate < 1 || .@rate > 99){
					set .@rate,90; //default;
				}
				next;
				break;
			case 3:
				next;
				mes "[Auto-Potion]";
				mes "กรุณาใส่ SP % ที่ต้องการให้ทำงาน";
				mes "";
				input .@rate,0,100;
				set AP_SP_RATE,.@rate;
				if(.@rate < 1 || .@rate > 99){
					set .@rate,90; //default;
				}
				next;
				break;
			case 4:
				next;
				mes "[Auto-Potion]";
				
				set .@menu$,"";
				for(set .@i,0 ; .@i < getarraysize($@AUTOPOT_HP) ; set .@i,.@i+1){
					if(countitem($@AUTOPOT_HP[.@i])){
						set .@menu$,.@menu$+""+getitemname($@AUTOPOT_HP[.@i])+":";
						set .@pass,1;
					}else{
						set .@menu$,.@menu$+":";
					}
				}
				set .@menu$,.@menu$+"ยกเลิก";
				
				if(!.@pass){
					mes "^FF0000ไม่พบไอเทมที่ใช้สำหรับเติม HP";
					next;
				}else{
					mes "กรุณาเลือกไอเทมที่ต้องการใช้สำหรับเติม HP";
					set .@index,select(.@menu$);
					if(.@index > getarraysize($@AUTOPOT_HP)){
						next;
					}else{
						set AP_HP_ITEMID,$@AUTOPOT_HP[.@index-1];
						mes "คุณได้เลือก "+getitemname(AP_HP_ITEMID);
						next;	
					}			
				}
				break;
			case 5:
				next;
				mes "[Auto-Potion]";
				
				set .@menu$,"";
				for(set .@i,0 ; .@i < getarraysize($@AUTOPOT_SP) ; set .@i,.@i+1){
					if(countitem($@AUTOPOT_SP[.@i])){
						set .@menu$,.@menu$+""+getitemname($@AUTOPOT_SP[.@i])+":";
						set .@pass,1;
					}else{
						set .@menu$,.@menu$+":";
					}
				}
				set .@menu$,.@menu$+"ยกเลิก";
				
				if(!.@pass){
					mes "^FF0000ไม่พบไอเทมที่ใช้สำหรับเติม SP";
					next;
				}else{
					mes "กรุณาเลือกไอเทมที่ต้องการใช้สำหรับเติม SP";
					set .@index,select(.@menu$);
					if(.@index > getarraysize($@AUTOPOT_SP)){
						next;
					}else{
						set AP_SP_ITEMID,$@AUTOPOT_SP[.@index-1];
						mes "คุณได้เลือก "+getitemname(AP_SP_ITEMID);
						next;	
					}			
				}
				break;
			default:
				goto EXIT;
				break;		
		}
	}

EXIT:
	close2;
	return;
}

-	script	start_autopot	-1,{

OnStart:
	//if(Hp < 1){
	//	@autopotion_on = 0;
	//	end;
	//}
	
	if (hp == 0 || getstatus(SC_STUN2) || getstatus(SC_BERSERK) || getstatus(SC_SATURDAYNIGHTFEVER) || getstatus(SC_GRAVITATION) ||
		getstatus(SC_TRICKDEAD) || getstatus(SC_HIDING) || getstatus(SC__SHADOWFORM) || getstatus(SC__INVISIBILITY) ||
		getstatus(SC__MANHOLE) || getstatus(SC_KAGEHUMI) || getstatus(SC_HEAT_BARREL_AFTER) ||
		getstatus(SC_STONE) || getstatus(SC_FREEZE) || getstatus(SC_STUN) || getstatus(SC_SLEEP) ||
		getstatus(SC_CRYSTALIZE) || getstatus(SC_DEEPSLEEP) || getstatus(SC_HANDICAPSTATE_CRYSTALLIZATION) ||
		getstatus(SC_HANDICAPSTATE_LIGHTNINGSTRIKE) || getstatus(SC_HANDICAPSTATE_FROSTBITE) || getstatus(SC_KINGS_GRACE) || getstatus(SC_SUHIDE) || getstatus(HANDICAPSTATE_SWOONING))
			goto RESTART;
			
	if ( @autopotion_on && ((Hp && Hp * 100 / MaxHp < AP_HP_RATE) || ( Sp * 100 / MaxSP < AP_SP_RATE))) {
		set .@need_eff,0;
			
		if(Hp && ((Hp * 100 / MaxHp) < AP_HP_RATE)){
			if (!countitem(AP_HP_ITEMID)) {
				if (@autopot_hp_none <= gettimetick(2)) {
					@autopot_hp_none = gettimetick(2) + 5;
					dispbottom "[Auto-Potion] HP:ไม่พบ '" + getitemname(AP_HP_ITEMID) + "' ในตัวคุณ";
				}
			}else{
				if(!getiteminfo(AP_HP_ITEMID,2))
					delitem AP_HP_ITEMID,1;
				consumeitem AP_HP_ITEMID;
				set .@need_eff,1;
			}
		}
		
		if((Sp * 100 / MaxSp) < AP_SP_RATE){
			if (!countitem(AP_SP_ITEMID)) {
				if (@autopot_sp_none <= gettimetick(2)) {
					@autopot_sp_none = gettimetick(2) + 5;
					dispbottom "[Auto-Potion] SP:ไม่พบ '" + getitemname(AP_SP_ITEMID) + "' ในตัวคุณ";
				}
			}else{
				if(!getiteminfo(AP_SP_ITEMID,2))
					delitem AP_SP_ITEMID,1;
				consumeitem AP_SP_ITEMID;
				set .@need_eff,1;	
			}
		}
		//if(.@need_eff)	specialeffect2 EF_POTION1;
	}
RESTART:
	addtimer $@AUTOPOT_DELAY,"start_autopot::OnStart";
	end;
	
}
