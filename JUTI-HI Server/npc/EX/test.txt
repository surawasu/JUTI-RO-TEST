				mes "[Apprentice Craftsman]";
				mes "ขอโทษทีท่านมี  " +getitemname(35000)+ " ไม่เพียงพอ.";
				close;
			}
			set .@enc_ran,rand(1,100);
			delitem 35000,5;
			delequip .@part;
	if(.@safe) {goto sussed;}
	if (.@enc_ran < .@failrate){
		goto sussed;
		}else {
				specialeffect2 EF_PHARMACY_FAIL;
				mes "[Apprentice Craftsman]";
				mes "ว๊า Enchant ผิดพลาด.";
				mes "ไอเทมหาย!!.";
				close;
			}
			
			sussed:
			
			setarray .@level[0],4700,4710,4720,4730,4740,4750,4701,4711,4721,4731,4741,4751,4702,4712,4722,4732,4742,4752;
			if (.@safe) {delitem 7146,1;}
			set .@addpart, .@level[rand(getarraysize(.@level))];
			getitem3 .@itemid,1,1,.@ref,0,.@card0,.@card1,.@card2,.@addpart,.@opt_id,.@opt_value,.@opt_param;
			end;
		}
	}
	else {
		mes "[Apprentice Craftsman]";
		mes "ท่านมีชุดเกาะนี้ในตัวละครมากกว่า 1 ชิ้น หรือ ท่านไม่มีเลย!";
		mes "ถ้าท่านมี 1 ชิ้นแล้วกลับมาหาข้าได้นะ.";
		close;
	}
	end;
}
