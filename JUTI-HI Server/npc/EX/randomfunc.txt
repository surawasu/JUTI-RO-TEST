	setarray .hsb3[0],60055; // rare
	setarray .hsb4[0],64010,52201,52202; // superrare
	set .hsb0rand,rand( getarraysize(.hsb0) ); // 
	set .hsb1rand,rand( getarraysize(.hsb1) ); // 
	set .hsb2rand,rand( getarraysize(.hsb2) ); // 
	set .hsb3rand,rand( getarraysize(.hsb3) ); // 
	set .hsb4rand,rand( getarraysize(.hsb4) ); // 	
			
	.chance = rand(1,1000);		
	query_sql "UPDATE `char` SET `point_gacha`=(`point_gacha` + 1) WHERE `char_id`='"+getcharid(0)+"' LIMIT 1";		
	query_sql ("SELECT `point_gacha` FROM `char` WHERE `char_id` = '"+ getcharid(0) +"'", .@ref);
	announce "[ ระบบ ]  :  ผู้เล่น ["+strcharinfo(0)+"] เปิด HotSale-Box ไปแล้วทั้งหมด "+.@ref+" ครั้ง",bc_all,0xFFFF00;
	if (.chance < 10 ) {	//0.2 rare
		announce "[ ระบบ ]  :  ผู้เล่น ["+strcharinfo(0)+"] เปิด HotSale-Box ได้รับ "+getitemname(.hsb4[.hsb4rand])+" ",bc_all,0xFFFF00;
		getitem .hsb4[.hsb4rand],1;
		end;
	}
	else if (.chance < 211 ) {
		announce "[ ระบบ ]  :  ผู้เล่น ["+strcharinfo(0)+"] เปิด HotSale-Box ได้รับ "+getitemname(.hsb3[.hsb3rand])+" ",bc_all,0xFFFF00;
		getitem .hsb3[.hsb3rand],rand(1,3);
		end;
	}
	else if (.chance < 361 ) {
		announce "[ ระบบ ]  :  ผู้เล่น ["+strcharinfo(0)+"] เปิด HotSale-Box ได้รับ "+getitemname(.hsb1[.hsb1rand])+" ",bc_all,0xFFFF00;
		getitem .hsb1[.hsb1rand],1;
		end;
	}
	else {
		getitem .hsb0[.hsb0rand],rand(1,5);
		end; 
	}
	
end;
