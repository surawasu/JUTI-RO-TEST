-	script	bot_logoff	-1,{
OnPCLoginEvent:
OnPCLogoutEvent:
if(getstatus( SC_AUTOATTACK )){
	sc_end SC_AUTOATTACK;
	//message strcharinfo(0),"ปิดการใช้งาน Auto Bot";
	}
if(!getstatus( SC_AUTOATTACK ))	
setautoattack("7;0;0");

}

-	script	bot_at	-1,{
	function	__onoff	;
OnInit:
	bindatcmd "botony",strnpcinfo(3)+"::OnDo";
	bindatcmd "botoffy",strnpcinfo(3)+"::OnDo2";
	bindatcmd "botendy",strnpcinfo(3)+"::OnDo2";
	end;

OnDo:
	if(getstatus( SC_AUTOATTACK )){
	sc_end SC_AUTOATTACK;
	message strcharinfo(0),"ปิดการใช้งาน Auto Bot";
	}
	if(autobot_time2 < gettimetick(2)){
	mes "ระยะเวลาบอทของท่านหมดลงแล้วโปรด\nทำการซื้อไอเท็ม<ITEM>"+getitemname(2832300)+"<INFO>2832300</INFO></ITEM> ก่อน";
	end;
	}
	
	.@AA_NA = 1;
	//.@duration = 3600000; // 1000 = 1s -- 60000 = 1m -- 3600000 = 1h etc...
	.@duration = (autobot_time2-gettimetick(2))*1000; // 1000 = 1s -- 60000 = 1m -- 3600000 = 1h etc...
	
	goto(main_menu);
	end;

main_menu:
	.@menu$ = "[ ตั้งค่าระบบออโต้บอท ]";
	mes .@menu$;
	
	.@main_menu = select("[เริ่มทำการบอท]","[ตั้งค่าบอท]","[ยกเลิก]");
	if( .@main_menu == 1 ){
		sc_start4 SC_AUTOATTACK,.@duration,.@AA_NA,0,0,0;
		message strcharinfo(0),"เปิดการใช้งาน Auto Bot";
	}
	if( .@main_menu == 2 )
		goto(config_menu);
	
	close;

config_menu:
	clear;
	.@menu$ = "[ ตั้งค่าระบบออโต้บอท ]";
	mes .@menu$;

	// To disable an option, comment the line
	menu	"Return",main_menu,
			" ~ ปิด-เปิด โจมตีปกติ",melee_attack,
			" ~ ตั้งค่าการฟื้นฟู",regeneration_menu,
			" ~ ตั้งค่า Skill โจมตี",attackskills_menu,
			" ~ ตั้งค่า Skill Buff",buffskills_menu,
			" ~ ตั้งค่า Buff Item",buffitems_menu,
			" ~ ตั้งค่า Teleport",teleport_menu,
			" ~ ตั้งค่า Monsters",monsters_menu,
			" ~ ตั้งค่า การเก็บ Item",pickupitems_menu;
	end;

pickupitems_menu:
	clear;
	.@menu$ = "[ตั้งค่า การเก็บ Item]";

	mes .@menu$;
	mes "--- การตั้งค่าปัจจุบัน ---";
	mes getautoattackstring(9,0);

	.@pickupitems_menu = select("ตั้งค่าการเก็บ","เพิ่มรายการไอเท็ม","ลบรายการไอเท็ม","ล้างรายการไอเท็มทั้งหมด","[Cancel]");

	switch(.@pickupitems_menu){
		case 1:
			.@pickup_config_choice = 0;
			// every item = 0
			// on the list = 1
			// no item = 2
			switch(getautoattackint(9,2)){
				case 0:
					.@pickupitems_config_menu  = select("เลือกเก็บไอเท็มเฉพาะที่ตั้งค่าไว้","ไม่เก็บไอเท็มที่ตั้งค่าไว้","[Cancel]");
					switch(.@pickupitems_config_menu){
						case 1:
							.@pickup_config_choice = 1;
							break;
						case 2 :
							.@pickup_config_choice = 2;
							break;
						case 3 :
							goto(pickupitems_menu);
							break;
					}
						
					break;
				case 1:
					.@pickupitems_config_menu  = select("เก็บไอเท็มทั้งหมด","ไม่เก็บไอเท็มทั้งหมด","[Cancel]");
					switch(.@pickupitems_config_menu){
						case 1:
							.@pickup_config_choice = 0;
							break;
						case 2 :
							.@pickup_config_choice = 2;
							break;
						case 3 :
							goto(pickupitems_menu);
							break;
					}
					break;
				case 2:
					.@pickupitems_config_menu  = select("เก็บไอเท็มทั้งหมด","เลือกเก็บไอเท็มเฉพาะที่ตั้งค่าไว้","[Cancel]");
					switch(.@pickupitems_config_menu){
						case 1:
							.@pickup_config_choice = 0;
							break;
						case 2 :
							.@pickup_config_choice = 1;
							break;
						case 3 :
							goto(pickupitems_menu);
							break;
					}
					break;				
			}
			setautoattack("9;2;" + .@pickup_config_choice);
			break;
		case 2:
			clear;
			mes .@menu$;
			mes "ใส่รหัส Item ที่ต้องการเพิ่ม ^FF0000(สูงสุด 20 รายการ)^000000";
			do {
				input .@itempickup_id;
			} while(.@itempickup_id < 1);

			setautoattack("9;1;" + .@itempickup_id);

			break;
		case 3:
			if(getautoattackint(9,1) > 0){
				mes "ใส่รหัส Item ที่ต้องการลบ";

				do {
					input .@itempickup_id;
				} while(.@itempickup_id < 1);
				setautoattack("9;0;" + .@itempickup_id);
			} else {
				clear;
				mes .@menu$;
				mes "คุณไม่มีไอเท็มนี้ในรายการ";
				next;
			}

			break;
		case 4:
			setautoattack("9;-1;0");
			break;

		case 5:
			goto(config_menu);
			break;
	}

	goto(pickupitems_menu);
	end;

monsters_menu:
	clear;
	.@menu$ = "[ตั้งค่า Monsters]";

	mes .@menu$;
	mes "--- การตั้งค่าปัจจุบัน ---";
	mes getautoattackstring(8,0);

	.@monsters_menu = select("เพิ่ม Monster","ลบ Monster","ล้างรายการทั้งหมด",getautoattackint(8,0)?"โจมตี Aggressive monsters":"ไม่สนใจ Aggressive monsters","[Cancel]");

	switch(.@monsters_menu){
		case 1:
			clear;
			mes .@menu$;
			mes "ใส่รหัส Monster ที่ต้องการเพิ่ม ^FF0000(สูงสุด 20 รายการ)^000000";
			do {
				input .@mob_id;
			} while(.@mob_id < 1);

			setautoattack("8;1;" + .@mob_id + ";" + getautoattackint(8,0));

			break;
		case 2:
			if(getautoattackint(8,1) > 0){
				mes "ใส่รหัส Monster ที่ต้องการลบ";

				do {
					input .@mob_id;
				} while(.@mob_id < 1);
				setautoattack("8;0;" + .@mob_id + ";" + getautoattackint(8,0));
			} else {
				clear;
				mes .@menu$;
				mes "คุณไม่มี Monster นี้ในรายการ";
				next;
			}

			break;
		case 3:
			setautoattack("8;-1;0;0");
			break;

		case 4:
			if(getautoattackint(8,0))
				setautoattack("8;0;0;0");
			else
				setautoattack("8;0;0;1");

			break;

		case 5:
			goto(config_menu);
			break;
	}

	goto(monsters_menu);
	end;

teleport_menu:
	clear;
	.@menu$ = "[Teleport]";

	mes .@menu$;
	mes "--- การตั้งค่าปัจจุบัน ---";
	mes getautoattackstring(7,0);

	if(getautoattackint(7,0) == 1)
		.@teleport_menu_txt$ = "เปิดใช้งาน Teleport Skill";
	else
		.@teleport_menu_txt$ = "ปิดใช้งาน Teleport Skill";

	if(getautoattackint(7,1) == 1)
		.@teleport_menu_txt$ += ":เปิดใช้งาน Fly Wing";
	else
		.@teleport_menu_txt$ += ":ปิดใช้งาน Fly Wing";
		
	.@teleport_menu_txt$ += ":ตั้งค่า/ปรับแต่ง Teleport เมื่อ HP ถึงที่กำหนด:ตั้งค่า/ปรับแต่ง Teleport Delay:[Cancel]";

	.@teleport_menu = select(.@teleport_menu_txt$);

	switch(.@teleport_menu){
		case 1:
			if(getautoattackint(7,0))
				setautoattack("7;0;0");
			else
				setautoattack("7;0;1");
			break;
		case 2:
			if(getautoattackint(7,1))
				setautoattack("7;1;0");
			else
				setautoattack("7;1;1");
			break;
		case 3:
			do {
				clear;
				mes .@menu$;
				mes "กำหนด HP% ที่ต้องการให้บอท Teleport หลบหนี";
				mes "ต่ำสุด 5 และ สูงสุด 90 \n(0 ปิดการใช้งาน)";
				input .@teleport_minhpvalue;
			} while(.@teleport_minhpvalue != 0 && (.@teleport_minhpvalue < 5 || .@teleport_minhpvalue > 90));
			setautoattack("7;2;"+.@teleport_minhpvalue);
			break;
		case 4:
			do {
				clear;
				mes .@menu$;
				mes "ตั่งค่าระยะเวลาที่ให้บอท Teleport \nเมื่อไม่เจอ Monster";
				mes "ต่ำสุด 2 และสูงสุด 600 วินาที\n(0 ปิดการใช้งาน)";
				input .@teleport_delayvalue;
			} while(.@teleport_delayvalue != 0 && (.@teleport_delayvalue < 2 || .@teleport_delayvalue > 600));
			setautoattack("7;3;"+.@teleport_delayvalue);
			break;
		default:
			goto(config_menu);
			break;
	}

	goto(teleport_menu);
	end;

melee_attack:
	clear;
	.@menu$ = "[ตั้งค่าโจมตีปกติ]";

	mes .@menu$;
	mes "--- การตั้งค่าปัจจุบัน ---";
	mes getautoattackstring(6,0);

	.@melee_menu = select(getautoattackint(6,0)?"เปิด":"ปิด","[Cancel]");

	if( .@melee_menu == 1 ){
		if(getautoattackint(6,0))
			setautoattack("6;0");
		else
			setautoattack("6;1");
		clear;

		goto(melee_attack);
	}

	goto(config_menu);
	end;

regeneration_menu:
	mes "Clear menu";
	clear;
	.@menu$ = "[ตั้งค่าการฟื้นฟู]";
	
	mes .@menu$;
	menu	"Return",config_menu,
			" ~ Auto-Heal",heal_menu,
			" ~ HP / SP Potions",potion_menu,
			" ~ นั่ง",sit_regen_menu;
	end;

heal_menu:
	clear;
	.@menu$ = "[Auto-Heal]";
	if(!getskilllv(28) && !getskilllv(2051) && !getskilllv(231)){
		mes .@menu$;
		mes "ขออภัยคุณไม่มี Skillในรายการด้านล่าง";
		mes "Heal, Highness Heal, Potion Pitcher";
		next;
		goto(regeneration_menu);
	}

	mes .@menu$;
	mes "--- การตั้งค่าปัจจุบัน ---";
	mes getautoattackstring(0,0);

	.@heal_menu = select("เพิ่ม / ปรับแต่ง skills","ปิดการใช้งาน skill","[Cancel]");
	if( .@heal_menu == 1 ){
		clear;
		mes .@menu$;
		mes "ใส่เลขSkillที่ต้องการเลือก";
		if(getskilllv(28)) // Heal
			mes "^FF0000(28)^000000 [" + getskillinfostring(0,28) + "]" + " - Max lv : " + getskilllv(28);
		if(getskilllv(2051)) // Highness Heal
			mes "^FF0000(2051)^000000 [" + getskillinfostring(0,2051) + "]" + " - Max lv : " + getskilllv(2051);
		if(getskilllv(231)) // Potion Pitcher
			mes "^FF0000(231)^000000 [" + getskillinfostring(0,231) + "]" + " - Max lv : " + getskilllv(231);

		input .@heal_skillid;
		if(.@heal_skillid < 1 )
			goto(heal_menu);

		if((.@heal_skillid != 28 && .@heal_skillid != 2051 && .@heal_skillid != 231) || !getskilllv(.@heal_skillid)){
			clear;
			mes .@menu$;
			mes "โปรดเลือกSkillที่มีอยู่";
			goto(heal_menu);
		} else {
			do{
				clear;
				mes .@menu$;
				mes "ระบุ Level Skill";
				mes "[" + getskillinfostring(0,.@heal_skillid) + "]" + " - Max lv : " + getskilllv(.@heal_skillid);
				input .@heal_skilllv;
			} while(.@heal_skilllv < 1 || .@heal_skilllv > getskilllv(.@heal_skillid));

			do {
				clear;
				mes .@menu$;
				mes "ระบุ HP% ที่ต้องการให้บอทให้Skill \n(เช่น : 10 บอทจะใช้เมื่อ HP ต่ำกว่า 10%)";
				mes "ต่ำสุด 5 และสูงสุด 90 %";
				mes "[" + getskillinfostring(0,.@heal_skillid) + "]" + " - Lv : " + .@heal_skilllv;
				input .@heal_minhpvalue;
			} while(.@heal_minhpvalue < 5 || .@heal_minhpvalue > 90);
			setautoattack("0;1;"+ .@heal_skillid + ";" + .@heal_skilllv + ";" + .@heal_minhpvalue);
			clear;
		}
		goto(heal_menu);
	}
	if( .@heal_menu == 2 ){
		if(getautoattackint(0,0) == 0){
			clear;
			mes .@menu$;
			mes "ไม่มี Skill ที่ต้องการปิดการใช้งาน";
			next;
			goto(heal_menu);
		} else {
			mes "ระบุเลขSkillที่ต้องการปิดการใข้งาน.";
			input .@heal_skillid;
			if(.@heal_skillid < 1 )
				goto(heal_menu);

			if((.@heal_skillid != 28 && .@heal_skillid != 2051 && .@heal_skillid != 231) || !getskilllv(.@heal_skillid)){
				clear;
				mes .@menu$;
				mes "โปรดเลือกSkillที่มีอยู่";
				goto(heal_menu);
			} else 
				setautoattack("0;0;"+ .@heal_skillid + ";0;0");
			goto(heal_menu);
		}
	}

	goto(regeneration_menu);
	end;

potion_menu:
	clear;
	.@menu$ = "[HP / SP Potions]";
	if(getautoattackint(1,0) == 0){
		mes .@menu$;
		mes "ขออภัยคุณไม่มี Item ฟื้นฟูอยู่";
		next;
		goto(regeneration_menu);
	}

	mes .@menu$;
	mes "--- การตั้งค่าปัจจุบัน ---";
	mes getautoattackstring(1,0);

	.@potion_menu = select("เพิ่ม / ปรับแต่ง Potions","ปิดการใช้งาน potion","[Cancel]");
	if( .@potion_menu == 1 ){
		clear;
		mes .@menu$;
		mes getautoattackstring(1,-1);
		mes "ระบุเลขไอเท็มที่ต้องการ";

		input .@potion_id;

		do {
			clear;
			mes .@menu$;
			mes "ระบบ HP% ที่ต้องการให้บอทใช้งาน Potion \n(เช่น : 10 คือเมื่อ HP ต่ำกว่า 10%)";
			mes "ต่ำสุด 5 และสูงสุด 90 \n(0 ปิดการใช้งาน)";
			input .@potion_minhpvalue;
		} while(.@potion_minhpvalue != 0 && (.@potion_minhpvalue < 5 || .@potion_minhpvalue > 90));

		do {
			clear;
			mes .@menu$;
			mes "ระบบ SP% ที่ต้องการให้บอทใช้งาน Potion \n(เช่น : 10 คือเมื่อ SP ต่ำกว่า 10%)";
			mes "ต่ำสุด 5 และสูงสุด 90 \n(0 ปิดการใช้งาน)";
			input .@potion_minspvalue;
		} while(.@potion_minspvalue != 0 && (.@potion_minspvalue < 5 || .@potion_minspvalue > 90));

		setautoattack("1;1;"+ .@potion_id + ";" + .@potion_minhpvalue + ";" + .@potion_minspvalue);
		clear;

		goto(potion_menu);
	}
	if( .@potion_menu == 2 ){
		if(getautoattackint(1,0) == 0){
			clear;
			mes .@menu$;
			mes "ไม่มีไอเท็มที่ต้องการลบ";
			next;
			goto(potion_menu);
		} else {
			mes "ระบุเลขไอเท็มที่ต้องการลบ.";
			input .@potion_id;
			setautoattack("1;0;"+ .@potion_id + ";0;0");
			goto(potion_menu);
		}
	}

	goto(regeneration_menu);
	end;

sit_regen_menu:
	clear;
	.@menu$ = "[นั่งเพื่อฟื้นฟู]";

	mes .@menu$;
	mes "--- การตั้งค่าปัจจุบัน ---";
	mes getautoattackstring(2,0); // status
	mes getautoattackstring(2,1); // hp
	mes getautoattackstring(2,2); // sp

	.@sit_regen_menu = select(getautoattackint(2,0)?"ปิด:ปรับแต่ง":"เปิด","[Cancel]");
	if( (.@sit_regen_menu == 1 && !getautoattackint(2,0))
		|| (.@sit_regen_menu == 2 && getautoattackint(2,0))){
		clear;
		mes .@menu$;
		mes "ระบบ HP% ที่ต้องการให้บอทนั่ง \n(เช่น : 10 คือเมื่อ HP ต่ำกว่า 10%)";
		mes "ต่ำสุด 5 และสูงสุด 90 \n(0 ปิดการใช้งาน)";

		do {
			input .@sit_regen_minhpvalue;
		} while(.@sit_regen_minhpvalue != 0 && (.@sit_regen_minhpvalue < 5 || .@sit_regen_minhpvalue > 90));

		if(.@sit_regen_minhpvalue > 0){
			clear;
			mes .@menu$;
			mes "ระบบ HP% ที่ต้องการให้บอทนั่ง \n(เช่น : 10 คือเมื่อ HP ต่ำกว่า 10%)";
			mes "ต่ำสุด 5 และสูงสุด 90\n (0 ปิดการใช้งาน)";
			mes "ระบุ HP ที่ต้องการให้บอทยืนขึ้น : " + .@sit_regen_minhpvalue;

			do {
				input .@sit_regen_maxhpvalue;
			} while((.@sit_regen_maxhpvalue != 0 && (.@sit_regen_maxhpvalue < 5 || .@sit_regen_maxhpvalue > 90)) || (.@sit_regen_minhpvalue >= .@sit_regen_maxhpvalue));
		}

		clear;
		mes .@menu$;
		mes "ระบบ SP% ที่ต้องการให้บอทนั่ง \n(เช่น : 10 คือเมื่อ SP ต่ำกว่า 10%)";
		mes "ต่ำสุด 5 และสูงสุด 90 \n(0 ปิดการใช้งาน)";

		do {
			input .@sit_regen_minspvalue;
		} while(.@sit_regen_minspvalue != 0 && (.@sit_regen_minspvalue < 5 || .@sit_regen_minspvalue > 90));

		if(.@sit_regen_minspvalue > 0){
			clear;
			mes .@menu$;
			mes "ระบบ SP% ที่ต้องการให้บอทนั่ง \n(เช่น : 10 คือเมื่อ SP ต่ำกว่า 10%)";
			mes "ต่ำสุด 5 และสูงสุด 90 \n(0 ปิดการใช้งาน)";
			mes "ระบุ SP ที่ต้องการให้บอทยืนขึ้น : " + .@sit_regen_minspvalue;

			do {
				input .@sit_regen_maxspvalue;
			} while((.@sit_regen_maxspvalue != 0 && (.@sit_regen_maxspvalue < 5 || .@sit_regen_maxspvalue > 90)) || (.@sit_regen_minspvalue >= .@sit_regen_maxspvalue));
		}
		
		setautoattack("2;1;"+ .@sit_regen_minhpvalue + ";" + .@sit_regen_maxhpvalue + ";" + .@sit_regen_minspvalue + ";" + .@sit_regen_maxspvalue);

	} else if( .@sit_regen_menu == 1 && getautoattackint(2,0)){
		setautoattack("2;0;0;0;0;0");
	} else {
		clear;
		goto(regeneration_menu);
	}

	clear;
	goto(sit_regen_menu);
	end;

buffskills_menu:
	clear;
	.@menu$ = "[Buff Skills]";

	mes .@menu$;
	mes "--- การตั้งค่าปัจจุบัน ---";
	mes getautoattackstring(3,0);
	
	.@buffskills_menu = select("เพิ่ม Skill","ปรับแต่ง skills","ปิดการใช้งาน skill","[Cancel]");
