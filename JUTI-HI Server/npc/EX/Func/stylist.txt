like_game,145,152,6	script	แต่งตัว#custom_stylist	503,{
//if(BaseLevel < 100){
//mes "ต้องLevelมากกว่า 100 ถึงจะใช้บริการได้";
//end;
//}
if(checkmadogear()){
	setarray .@Styles[1],
		1,
		getbattleflag("max_hair_style"),
		getbattleflag("max_hair_color");
}
else if(Class == Job_Summoner){
	setarray .@Styles[1],
		3,
		10,
		1;
}else{
	setarray .@Styles[1],
		getbattleflag("max_cloth_color"),
		getbattleflag("max_hair_style"),
		getbattleflag("max_hair_color");
}
	setarray .@Look[1],
		LOOK_CLOTHES_COLOR,
		LOOK_HAIR,
		LOOK_HAIR_COLOR;
	set .@s, select(" ~ สีชุด: ~ ทรงผม: ~ สีผม");
	set .@Revert, getlook(.@Look[.@s]);
	set .@Style,1;
	while(1) {
		setlook .@Look[.@s], .@Style;
		message strcharinfo(0),"นี่คือสีที่ #"+.@Style+".";
		set .@menu$, " ~ สีต่อไป (^0055FF"+((.@Style!=.@Styles[.@s])?.@Style+1:1)+"^000000): ~ สีก่อนหน้า (^0055FF"+((.@Style!=1)?.@Style-1:.@Styles[.@s])+"^000000): ~ ไปที่สี...: ~ กลับไปสีเดิม (^0055FF"+.@Revert+"^000000)";
		switch(select(.@menu$)) {
		case 1:
			set .@Style, ((.@Style != .@Styles[.@s]) ? .@Style+1 : 1);
			break;
		case 2:
			set .@Style, ((.@Style != 1) ? .@Style-1 : .@Styles[.@s]);
			break;
		case 3:
			message strcharinfo(0),"โปรดระบุตัวเลขตั้งแต่ 1 - "+.@Styles[.@s]+".";
			input .@Style,0,.@Styles[.@s];
			if (!.@Style)
				set .@Style, rand(1,.@Styles[.@s]);
			break;
		case 4:
			set .@Style, .@Revert;
			setlook .@Look[.@s], .@Revert;
			break;
		}
	}
end;
OnInit:
waitingroom "เปลี่ยนสีชุด,ทรงผม",0;
end;
}
