morocc03,119,205,6	script	ถอดการ์ด 	402,{
	set .zenycost,5000000;    // ค่าธรรมเนียม ในการถอดการ์ด
	set .percardcost,1000000;  // ค่าธรรมเนียม ในการถอดการ์ดต่อหนึ่งใบ
	.rate = 100;
	//.req1 = 984;
	//.req2 = 985;
	//.amt1 = 10;
	//.amt2 = 10;
	
	
	
	disable_items;
	mes "[นักถอดการ์ด ]";
	mes "ข้าต้องการค่าธรรมเนียม "+callfunc("F_InsertComma",.zenycost)+" zeny, บวกกับ "+callfunc("F_InsertComma",.percardcost)+" zeny ต่อการ์ดหนึ่งใบ.";
	//mes "และ <ITEM>"+getitemname(.req1)+"<INFO>"+.req1+"</INFO></ITEM> "+.amt1+" ea ";
	//mes "<ITEM>"+getitemname(.req2)+"<INFO>"+.req2+"</INFO></ITEM> "+.amt2+" ea ";
	mes "โอกาสสำเร็จ "+(.rate)+"%";
	next;
	switch(select("ถอดการ์ด.:ถามค่าใช้จ่าย:ออก.")) {
	case 1:
		mes "[นักถอดการ์ด ]";
		mes "เลือกของสวมใส่ที่ต้องการถอดการ์ด";
		next;

		setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
		for( set .@i,1; .@i <= 10; set .@i,.@i+1 ) {
			if( getequipisequiped(.@indices[.@i]) )
				set .@menu$, .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			set .@menu$, .@menu$ + ":";
		}
		set .@part, .@indices[ select(.@menu$) ];
		if(!getequipisequiped(.@part)) {
			mes "[นักถอดการ์ด ]";
			mes "ไม่เห็นว่าเจ้าจะใส่อะไรมาเลย.";
			close;
		}
		if(getequipcardcnt(.@part) == 0) {
			mes "[นักถอดการ์ด ]";
			mes "ของชิ้นนี้ไม่มีการ์ด";
			close;
		}
		set .@cardcount,getequipcardcnt(.@part);
		
		if (!checkweight(1202,(.@cardcount+1))) { 
			mes "[นักถอดการ์ด ]";
			mes "เจ้ามีช่องว่างในกระเป๋าไม่พอ.";
			close;
		}
		mes "[นักถอดการ์ด ]";
		mes "ของชิ้นนี้มีการ์ด " + .@cardcount + " ใบ. เจ้าต้องจ่าย " + (.zenycost+(.@cardcount * .percardcost)) + " zeny สำหรับถอดการ์ด.";
		//mes "และ <ITEM>"+getitemname(.req1)+"<INFO>"+.req1+"</INFO></ITEM> "+.amt1+" ea ";
		//mes "<ITEM>"+getitemname(.req2)+"<INFO>"+.req2+"</INFO></ITEM> "+.amt2+" ea ";
		mes "โอกาสสำเร็จ "+(.rate)+"%";
		next;
		if(select("ลุยเลย.:ไม่มี.") == 2) {
			mes "[นักถอดการ์ด ]";
			mes "ยากจนจริงๆเชียว.";
			close;
		}
		if(zeny < (.zenycost+(.@cardcount * .percardcost))) {
			mes "[นักถอดการ์ด ]";
			mes "เจ้ามีเงินไม่พอ.";
			close;
		}
		/*
		if(countitem(.req1) < .amt1 || countitem(.req2) < .amt2){
			mes "[นักถอดการ์ด ]";
			mes "มีของที่ต้องการไม่เพียงพอ.";
			close;
		}
		*/
		next;
		.@rand = rand(0,99);
		set Zeny, Zeny - (.zenycost+(.@cardcount * .percardcost));
		//delitem .req1,.amt1;
		//delitem .req2,.amt2;
		if(.@rand < .rate){
			specialeffect2 66;
			successremovecards .@part;
			mes "[นักถอดการ์ด ]";
			mes "เรียบร้อยแล้ว.";
		close;
		}else{
			specialeffect2 585;
			mes "[นักถอดการ์ด ]";
			mes "แย่แล้วเหมือนจะล้มเหลวลองใหม่อีกครั้งละกันนะ.";
		close;
		}
	case 2:
		mes "[นักถอดการ์ด ]";
		mes "ข้าต้องการค่าธรรมเนียม "+callfunc("F_InsertComma",.zenycost)+" zeny, บวกกับ "+callfunc("F_InsertComma",.percardcost)+" zeny ต่อการ์ดหนึ่งใบ.";
		mes "และ <ITEM>"+getitemname(.req1)+"<INFO>"+.req1+"</INFO></ITEM> "+.amt1+" ea ";
		mes "<ITEM>"+getitemname(.req2)+"<INFO>"+.req2+"</INFO></ITEM> "+.amt2+" ea ";
		close;
	case 3:
		mes "[นักถอดการ์ด ]";
		mes "บ๊าย บาย.";
		close;
	}

end;
	
OnInit:
waitingroom "     ถอดการ์ด ",0;
end;		
	
}
