function	script	bacc_add_bet	{
	set .BetAmt, getarg(0);
	set .ItemId, getarg(1);
	if ($BetMode == 1) {
		set ZENY, ZENY + .BetAmt;
		dispbottom "ผู้เล่นได้รับ "+callfunc("F_InsertComma", .BetAmt)+" Zeny";
		dispbottom "ยอดเงินคงเหลือ "+callfunc("F_InsertComma", ZENY)+" Zeny";
	} else if ($BetMode == 2) {
		set #CASHPOINTS, #CASHPOINTS + .BetAmt;
		dispbottom "ผู้เล่นได้รับ "+callfunc("F_InsertComma", .BetAmt)+" Cash";
		dispbottom "ยอดเงินคงเหลือ "+callfunc("F_InsertComma", #CASHPOINTS)+" Zeny";
	} else if ($BetMode == 3) {
		if (getiteminfo(.ItemId,2) == 4 || getiteminfo(.ItemId,2) == 5) {
			for (set .@ibet1, 0; .@ibet1 < .BetAmt; set .@ibet1, .@ibet1 + 1)
				getitem .ItemId, 1;
		} else {
			getitem .ItemId, .BetAmt;
		}
		dispbottom "ผู้เล่นได้รับ "+getitemname($ItemID)+" จำนวน "+callfunc("F_InsertComma", .BetAmt)+" ชิ้น";
	}
	return;
}
